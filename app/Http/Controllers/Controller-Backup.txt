      public function inquirySubmit(Request $request)
    {
        // Validation
        $validatedData = $request->validate([
            'name' => 'required|regex:/^[a-zA-Z\s]+$/|max:30',
            'subject' => 'required',
            'message' => 'required|regex:/^[a-zA-Z\s]+$/|max:100',
            'phone' => 'required|regex:/^[0-9]{1,10}$/|max:10',
            'email' => 'required|email',
        ], [
            'name.required' => 'Please enter the name.',
            'name.regex' => 'Please enter a valid name.',
            'name.max' => 'Please enter maximum 30 characters.',
            'subject.required' => 'Please select the subject.',
            'message.required' => 'Please enter the message.',
            'message.regex' => 'Please enter a valid message.',
            'message.max' => 'Please enter maximum 100 characters.',
            'phone.required' => 'Please enter the contact number.',
            'phone.regex' => 'Please enter valid contact number.',
            'phone.max' => 'Please enter maximum 10 digits.',
            'email.required' => 'Please enter the email.',
            'email.email' => 'Please enter a valid email address.',
        ]);

        // Data to be sent to SheetDB
        $data = [
            'name' => $validatedData['name'],
            'email' => $validatedData['email'],
            'phone' => $validatedData['phone'],
            'subject' => $validatedData['subject'],
            'message' => $validatedData['message'],
             'source' => $request->input('source')
        ];
      
        // Send data to SheetDB
       try {
              $result = Contact::create($data);
          

            $client = new Client();
            $response = $client->post('https://www.ksquareenergy.cloud/web/hook/22a279e2-a907-4e4b-8b14-bf02ce7809f0', [
                'json' => [$data]
            ]);
//  dd($response->getStatusCode());
            // Debugging response
            $responseData = json_decode($response->getBody(), true);
            //dd($responseData);

            if ($response->getStatusCode() === 200) {
                return redirect('thank-you')->with('success', 'Your message has been submitted successfully!');
            } else {
                return redirect()->back()->with('error', 'Failed to submit your message. Please try again later.');
            }
        } catch (\Exception $e) {
            Log::error('Error submitting contact form: ' . $e->getMessage());
            return redirect()->back()->with('error', 'Failed to submit your message. Please try again later.');
        }
    }
